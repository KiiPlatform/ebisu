48#define KII_AUTHOR_ID_MAX_SIZE 128
49/* The sizes of the tokens, and of kii_mqtt_endpoint_t's fields below, can be
50 * defined otherwise, for a server that issues longer tokens or a broker that
51 * gives longer fields: alike for kii and everything using it, as they change
52 * the layout of kii_author_t, kii_mqtt_endpoint_t and kii_t. A session's
53 * renewal needs as much more room in the buffer set by kii_set_buff(), as its
54 * response carries the tokens, and, by onboarding again, the endpoint. A token
55 * longer than its size fails the login, onboarding or renewal that got it
56 * with KII_ERR_TOO_LARGE_DATA. */
